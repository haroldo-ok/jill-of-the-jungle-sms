# Jill of the Jungle SMS - automated playtest (run against jill.sms with its .noi symbols)
symbols jill.noi

echo [1] title screen shows the menu; button 1 starts the game
run 120
expectvdp display 1
expectcolors 3
screenshot shots/00_title.ppm
hold 1
run 3
hold none
run 200
expect cur_level == 0 w1
expect health == 6 w1
expectcolors 6
screenshot shots/01_hub.ppm

echo [2] walking right moves Jill
hold right
run 60
hold none
expect px > 100 s16

echo [2b] hub lift (object 9): stand on it, hold DOWN, it lowers one cell at a time and carries Jill
poke msg_shown 16 w1
poke px 960 w2
poke py 140 w2
run 60
expect oy+18 == 160 s16
hold down
run 60
hold none
run 10
expect oy+18 > 176 s16
expect py > 168 s16
hold up
run 120
hold none
run 10
expect oy+18 < 176 s16
poke px 112 w2
poke py 190 w2
run 120

echo [3] pause menu opens, Save game writes SRAM, resume
pause
run 60
screenshot shots/02_pause.ppm
hold down
run 3
hold none
run 8
hold 1
run 3
hold none
run 120
expect cur_level == 0 w1

echo [4] every level loads and keeps running (no reset, no crash)
poke req_level 1 w1
run 200
expect cur_level == 1 w1
screenshot shots/03_level1.ppm
poke req_level 2 w1
run 200
expect cur_level == 2 w1
poke req_level 3 w1
run 200
expect cur_level == 3 w1
poke req_level 4 w1
run 200
expect cur_level == 4 w1
poke req_level 5 w1
run 200
expect cur_level == 5 w1
poke req_level 6 w1
run 200
expect cur_level == 6 w1

echo [5] demo worlds (one-shot hint pages suppressed)
poke msg_shown 31 w1
poke req_level 8 w1
run 200
expect cur_level == 8 w1
poke req_level 9 w1
run 200
expect cur_level == 9 w1
poke req_level 10 w1
run 200
expect cur_level == 10 w1
screenshot shots/04_demo3.ppm

echo [6] ending: macro walk then story slides, then back to the title
poke req_level 7 w1
run 2600
screenshot shots/05_ending.ppm
